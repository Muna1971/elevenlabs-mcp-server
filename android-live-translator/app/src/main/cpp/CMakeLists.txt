cmake_minimum_required(VERSION 3.22.1)
project(nativecore C CXX)

set(CMAKE_CXX_STANDARD 17)
include(FetchContent)

# --- whisper.cpp: offline multilingual speech recognition + Silero voice detection ---
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# Same target as the official whisper.android example (phones from ~2018 onwards).
set(GGML_CPU_ARM_ARCH "armv8.2-a+fp16" CACHE STRING "" FORCE)

FetchContent_Declare(whisper
    URL https://github.com/ggml-org/whisper.cpp/archive/refs/tags/v1.8.7.tar.gz)
FetchContent_MakeAvailable(whisper)

# --- slimt (Mozilla Bergamot / Firefox Translations engine) via slimt-sys's C API ---
FetchContent_Declare(slimtsys
    GIT_REPOSITORY https://github.com/DavidVentura/slimt-sys.git
    GIT_TAG 1cef6a655b713aef25695eba8451d10ac1c4f21a)
FetchContent_GetProperties(slimtsys)
if(NOT slimtsys_POPULATED)
  FetchContent_Populate(slimtsys)
endif()
set(SLIMT_SOURCE_DIR ${slimtsys_SOURCE_DIR}/vendor/slimt)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(USE_NEON ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(${slimtsys_SOURCE_DIR}/bindings slimtsys)
# Headers generated by slimt land below our binary dir when it is a subproject.
target_include_directories(slimt-static PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/slimtsys/slimt)
target_include_directories(slimt_wrapper PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/slimtsys/slimt)
target_compile_options(slimt-static PRIVATE -Wno-error)

add_library(nativecore SHARED whisper_jni.cpp bergamot_jni.cpp)
target_compile_options(nativecore PRIVATE -O3)
target_link_libraries(nativecore PRIVATE whisper slimt_wrapper slimt-static log)
